## استفاده

```php
use Tavaan\Engine\Compat\Compat;
use Tavaan\Engine\Compat\Utf8;
use Tavaan\Engine\Compat\ZipWriterPure;
use Tavaan\Engine\Compat\MagicBytesDetector;

// بررسی افزونه‌ها
$compat = new Compat([]);
$compat->has('mbstring');              // bool
$compat->hasAny(['imagick', 'gd']);     // bool
$compat->require('gd', 'image.resize'); // throws EngineCapabilityException
$compat->driver('image');              // 'imagick'|'gd'|'none'
$compat->report();                     // full extension report

// کار با UTF-8 (با fallback خالص PHP در صورت نبود mbstring)
Utf8::length('سلام');                  // 5
Utf8::substr('hello', 1, 3);            // 'ell'
Utf8::split('سلام');                    // ['س', 'ل', 'ا', 'م']
Utf8::lower('HELLO');                   // 'hello'
Utf8::upper('hello');                   // 'HELLO'

// ZIP خالص PHP (بدون ext-zip)
$zip = new ZipWriterPure();
$zip->addFromString('hello.txt', 'Hello, World!');
$zip->saveTo('/tmp/test.zip');

// تشخیص MIME بدون ext-fileinfo
MagicBytesDetector::detect("\xFF\xD8\xFF...");  // 'image/jpeg'
MagicBytesDetector::detectFromFile('/tmp/image.jpg');
```
